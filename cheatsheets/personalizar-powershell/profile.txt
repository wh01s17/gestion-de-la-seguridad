function Invoke-Starship-PreCommand {
    $id = Get-History -Count 1 | ForEach-Object { $_.Id }
    if ($id -and $id -ne $global:StarshipLastHistoryId) { Write-Host }
    $global:StarshipLastHistoryId = $id
}
